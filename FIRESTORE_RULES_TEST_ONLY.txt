rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // TESTING ONLY — the current app has no Firebase Authentication yet.
    // Do not leave these rules open on a public production site.
    match /framefusion_crew/{document=**} {
      allow read, write: if true;
    }

    match /framefusion_projects/{document=**} {
      allow read, write: if true;
    }

    match /framefusion_signatures/{document=**} {
      allow read, write: if true;
    }

    match /framefusion_receipts/{document=**} {
      allow read, write: if true;
    }

    match /framefusion_settings/{document=**} {
      allow read, write: if true;
    }

    // Needed by Firebase Trigger Email.
    // WARNING: public create access can be abused. Use only for testing.
    match /mail/{documentId} {
      allow create: if true;
      allow read, update, delete: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
