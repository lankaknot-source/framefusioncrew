rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function profilePath() {
      return /databases/$(database)/documents/framefusion_users/$(request.auth.uid);
    }

    function hasProfile() {
      return signedIn() && exists(profilePath());
    }

    function profile() {
      return get(profilePath()).data;
    }

    function validManagementRole() {
      return hasProfile() && (
        profile().role == "admin" ||
        profile().role == "director" ||
        profile().role == "manager" ||
        profile().role == "accountant"
      );
    }

    function activeUser() {
      return validManagementRole() && profile().active != false;
    }

    function isAdmin() {
      return activeUser() && profile().role == "admin";
    }

    function isOperationsManager() {
      return activeUser() && (
        profile().role == "admin" ||
        profile().role == "director" ||
        profile().role == "manager"
      );
    }

    function isFinance() {
      return activeUser() && (
        profile().role == "admin" ||
        profile().role == "director" ||
        profile().role == "manager" ||
        profile().role == "accountant"
      );
    }

    match /framefusion_users/{userId} {
      allow read: if activeUser() && (request.auth.uid == userId || isAdmin());
      allow create, update, delete: if isAdmin();
    }

    match /framefusion_tasks/{taskId} {
      allow read: if activeUser();
      allow create, update, delete: if isOperationsManager();
    }

    match /framefusion_crew/{document=**} {
      allow read: if activeUser();
      allow write: if isOperationsManager();
    }

    match /framefusion_projects/{document=**} {
      allow read: if activeUser();
      allow write: if isFinance();
    }

    match /framefusion_signatures/{document=**} {
      allow read: if activeUser();
      allow write: if isOperationsManager();
    }

    match /framefusion_receipts/{document=**} {
      allow read: if activeUser();
      allow write: if isFinance();
    }

    match /framefusion_rentals/{document=**} {
      allow read: if activeUser();
      allow write: if isFinance();
    }

    match /framefusion_settings/{document=**} {
      allow read: if activeUser();
      allow write: if isAdmin();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
